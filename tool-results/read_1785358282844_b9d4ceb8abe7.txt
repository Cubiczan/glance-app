    99→def wrap_text(text, font, max_width, draw):
   100→    words = text.split()
   101→    lines = []
   102→    current = ""
   103→    for word in words:
   104→        test = f"{current} {word}".strip()
   105→        bbox = draw.textbbox((0, 0), test, font=font)
   106→        if bbox[2] - bbox[0] <= max_width:
   107→            current = test
   108→        else:
   109→            if current:
   110→                lines.append(current)
   111→            current = word
   112→    if current:
   113→        lines.append(current)
   114→    return lines

   595→def assemble_video(slides):
   596→    """Use FFMPEG to assemble slides into MP4 with transitions."""
   597→    # Create individual segment videos with fade in/out
   598→    segments = []
   599→    for i, (path, duration) in enumerate(slides):
   600→        seg_path = os.path.join(FRAME_DIR, f"seg_{i:03d}.mp4")
   601→        fade_dur = 0.5
   602→        cmd = [
   603→            "ffmpeg", "-y", "-loop", "1", "-i", path,
   604→            "-t", str(duration),
   605→            "-vf", f"scale=1920:1080,fade=t=in:st=0:d={fade_dur},fade=t=out:st={duration-fade_dur}:d={fade_dur}",
   606→            "-c:v", "libx264", "-preset", "medium", "-crf", "23",
   607→            "-pix_fmt", "yuv420p", "-r", str(FPS),
   608→            seg_path
   609→        ]
   610→        subprocess.run(cmd, capture_output=True, check=True)
   611→        segments.append(seg_path)
   612→        print(f"  Segment {i}: {duration}s")
   613→
   614→    # Concatenate all segments
   615→    concat_file = os.path.join(FRAME_DIR, "concat.txt")
   616→    with open(concat_file, "w") as f:
   617→        for seg in segments:
   618→            f.write(f"file '{seg}'\n")
   619→
   620→    cmd = [
   621→        "ffmpeg", "-y", "-f", "concat", "-safe", "0", "-i", concat_file,
   622→        "-c:v", "libx264", "-preset", "medium", "-crf", "20",
   623→        "-pix_fmt", "yuv420p", "-movflags", "+faststart",
   624→        OUTPUT
   625→    ]
   626→    result = subprocess.run(cmd, capture_output=True, text=True)
   627→    if result.returncode != 0:
   628→        print(f"FFMPEG error: {result.stderr[-500:]}")
   629→        return False
   630→
   631→    # Get file size
   632→    size_mb = os.path.getsize(OUTPUT) / (1024 * 1024)
   633→    print(f"\n  Video saved: {OUTPUT}")
   634→    print(f"  Size: {size_mb:.1f} MB")
   635→
   636→    # Get duration
   637→    probe = subprocess.run(
   638→        ["ffprobe", "-v", "error", "-show_entries", "format=duration", "-of", "default=noprint_wrappers=1:nokey=1", OUTPUT],
   639→        capture_output=True, text=True
   640→    )
   641→    if probe.stdout.strip():
   642→        print(f"  Duration: {float(probe.stdout.strip()):.1f}s")
   643→    return True
